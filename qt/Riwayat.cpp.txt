Riwayat::Riwayat(QWidget *parent) : QWidget(parent)
{
    setStyleSheet("background:#f8fafc;");
    buildUI();
}

void Riwayat::buildUI()
{
    QVBoxLayout *pg = new QVBoxLayout(this);
    pg->setContentsMargins(32,28,32,28);
    pg->setSpacing(0);

    // Header: judul + search bar
    QHBoxLayout *hdr  = new QHBoxLayout;
    QVBoxLayout *hdrL = new QVBoxLayout; hdrL->setSpacing(4);

    QLabel *title = new QLabel("Riwayat Perhitungan");
    title->setStyleSheet("font-size:26px;font-weight:bold;color:#0f172a;background:transparent;");
    QLabel *sub = new QLabel("");
    sub->setStyleSheet("font-size:13px;color:#64748b;background:transparent;");
    hdrL->addWidget(title); hdrL->addWidget(sub);

    searchInput = new QLineEdit;
    searchInput->setPlaceholderText("Cari nama atau nomor polisi...");
    searchInput->setStyleSheet(R"(
        QLineEdit {
            background:#fff; border:1.5px solid #e2e8f0;
            border-radius:20px; padding:10px 16px;
            font-size:13px; color:#374151;
        }
        QLineEdit:focus { border-color:#2563eb; }
    )");
    searchInput->setFixedHeight(40);
    searchInput->setMinimumWidth(280);
    connect(searchInput, &QLineEdit::textChanged, this, &Riwayat::onSearch);

    hdr->addLayout(hdrL);
    hdr->addStretch();
    hdr->addWidget(searchInput);
    pg->addLayout(hdr);
    pg->addSpacing(20);

    // Tabel riwayat
    tabel = new QTableWidget(0, 7);
    tabel->setStyleSheet(SS_TABLE);
    tabel->setHorizontalHeaderLabels(
        {"No","Nama Pemilik","Nomor Polisi","Jenis Kendaraan",
         "NJKB (Rp)","Total Pajak (Rp)","Tanggal"});
    tabel->horizontalHeader()->setSectionResizeMode(QHeaderView::Stretch);
    tabel->horizontalHeader()->setSectionResizeMode(0, QHeaderView::Fixed);
    tabel->setColumnWidth(0, 50);
    tabel->verticalHeader()->setVisible(false);
    tabel->setEditTriggers(QAbstractItemView::NoEditTriggers);
    tabel->setSelectionBehavior(QAbstractItemView::SelectRows);
    tabel->setShowGrid(false);
    tabel->setFocusPolicy(Qt::NoFocus);
    tabel->verticalHeader()->setDefaultSectionSize(52);
    tabel->setCursor(Qt::PointingHandCursor);

    // Hubungkan klik sel ke slot
    connect(tabel, &QTableWidget::cellClicked, this, &Riwayat::onCellClicked);

    pg->addWidget(tabel, 1);
}

// Mengisi ulang tabel dari data terbaru. Dipanggil setiap kali halaman ini ditampilkan.
void Riwayat::refresh(const QList<DataKendaraan> &daftar)
{
    indexMapping.clear();
    tabel->setRowCount(0);

    for (int i = 0; i < daftar.size(); i++) {
        const DataKendaraan &d = daftar[i];
        int row = tabel->rowCount();
        tabel->insertRow(row);
        indexMapping.append(i); // simpan index asli untuk mapping

        auto addCell = [&](int col, const QString &val, bool blue = false) {
            QTableWidgetItem *it = new QTableWidgetItem(val);
            it->setTextAlignment(Qt::AlignVCenter | Qt::AlignLeft);
            if (blue) it->setForeground(QColor(37, 99, 235));
            tabel->setItem(row, col, it);
        };

        addCell(0, QString::number(i + 1));
        addCell(1, d.namaPemilik);
        addCell(2, d.nomorPolisi);
        addCell(3, d.jenisKendaraan);
        addCell(4, Save::formatRupiah(d.njkb));
        addCell(5, Save::formatRupiah(d.totalPajak), true);
        addCell(6, d.tanggalHitung);
    }
}

// Filter baris tabel berdasarkan teks pencarian (nama atau nomor polisi).
void Riwayat::onSearch(const QString &text)
{
    for (int i = 0; i < tabel->rowCount(); i++) {
        bool hit = text.isEmpty();
        if (!hit) {
            QString nm  = tabel->item(i,1) ? tabel->item(i,1)->text() : "";
            QString npl = tabel->item(i,2) ? tabel->item(i,2)->text() : "";
            hit = nm.contains(text, Qt::CaseInsensitive)
                  || npl.contains(text, Qt::CaseInsensitive);
        }
        tabel->setRowHidden(i, !hit);
    }
}

// Saat baris diklik, pancarkan sinyal dengan index asli data.
void Riwayat::onCellClicked(int row, int /*column*/)
{
    if (row >= 0 && row < indexMapping.size())
        emit barisDiklik(indexMapping[row]);
}
