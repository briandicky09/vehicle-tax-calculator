// ── Helper lokal

static QFrame* makeCard() {
    QFrame *f = new QFrame;
    f->setStyleSheet("QFrame{background:#ffffff;border:1px solid #e2e8f0;border-radius:12px;}");
    auto *sh = new QGraphicsDropShadowEffect;
    sh->setBlurRadius(12);
    sh->setColor(QColor(0, 0, 0, 15));
    sh->setOffset(0, 2);
    f->setGraphicsEffect(sh);
    return f;
}

/**
 * @brief makeLine
 * Membuat garis horizontal tipis sebagai pemisah antar section dalam card.
 */
static QFrame* makeLine() {
    QFrame *f = new QFrame;
    f->setFrameShape(QFrame::HLine);
    f->setFixedHeight(1);
    f->setStyleSheet("background:#e2e8f0;border:none;");
    return f;
}

/**
 * @brief makeCaption
 * Membuat QLabel kecil berwarna abu-abu untuk label field formulir.
 * @param t Teks label yang akan ditampilkan (biasanya huruf kapital semua).
 */
static QLabel* makeCaption(const QString &t) {
    QLabel *l = new QLabel(t);
    l->setStyleSheet("font-size:11px;font-weight:bold;color:#64748b;"
                     "letter-spacing:0.8px;background:transparent;border:none;");
    return l;
}

// ── MainWindow ────────────────────────────────────────────────

/**
 * @brief MainWindow::MainWindow
 * Konstruktor utama. Inisialisasi UI, lalu muat data riwayat dari file
 * agar data tetap ada meskipun aplikasi sudah ditutup sebelumnya.
 */
MainWindow::MainWindow(QWidget *parent)
    : QMainWindow(parent)
{
    setWindowTitle("Sistem Pajak Kendaraan Bermotor");
    setMinimumSize(1100, 700);
    resize(1200, 750);
    setupUI();

    // Muat data riwayat dari file saat aplikasi pertama kali dibuka
    riwayatData = Save::muatDariFile();

    showInputPage();
}

MainWindow::~MainWindow() {}

/**
 * @brief MainWindow::setupUI
 * Menyusun layout utama aplikasi: sidebar di kiri dan contentStack di kanan.
 * Semua halaman (Input, Hasil, Riwayat, Detail) didaftarkan ke contentStack
 * dan sinyal antar widget dikoneksikan di sini.
 */
void MainWindow::setupUI()
{
    centralWidget = new QWidget(this);
    setCentralWidget(centralWidget);

    mainLayout = new QHBoxLayout(centralWidget);
    mainLayout->setContentsMargins(0, 0, 0, 0);
    mainLayout->setSpacing(0);

    setupSidebar();

    contentStack = new QStackedWidget;
    contentStack->setStyleSheet("QStackedWidget{background:#f8fafc;}");

    // Bangun masing-masing halaman
    setupInputPage();

    hasilPage   = new Hasil(this);
    riwayatPage = new Riwayat(this);
    detailPage  = new Detail(this);

    // Hubungkan sinyal dari halaman Hasil ke slot MainWindow
    connect(hasilPage, &Hasil::kembaliDiminta, this, &MainWindow::kembaliKeInput);
    connect(hasilPage, &Hasil::simpanDiminta,  this, &MainWindow::simpanKeRiwayat);

    // Hubungkan sinyal dari halaman Riwayat: saat baris diklik buka Detail
    connect(riwayatPage, &Riwayat::barisDiklik, this, &MainWindow::bukaDetailRiwayat);

    // Hubungkan sinyal dari halaman Detail ke slot MainWindow
    connect(detailPage, &Detail::kembaliDiminta, this, &MainWindow::kembaliKeRiwayatDariDetail);
    connect(detailPage, &Detail::hapusDiminta,   this, &MainWindow::hapusRiwayat);
    connect(detailPage, &Detail::editDisimpan,   this, &MainWindow::updateRiwayat);

    // Daftarkan semua halaman ke stack
    contentStack->addWidget(inputPage);
    contentStack->addWidget(hasilPage);
    contentStack->addWidget(riwayatPage);
    contentStack->addWidget(detailPage);

    mainLayout->addWidget(sidebar);
    mainLayout->addWidget(contentStack, 1);
}

/**

 * Membangun halaman formulir Input Data Kendaraan.

 * saat ukuran jendela lebih kecil dari total tinggi formulir.
 * Berisi dua card: Informasi Pemilik dan Data Kendaraan.

    pg->setContentsMargins(32, 28, 32, 28);

    pg->addWidget(sub);

    pL->setContentsMargins(28, 20, 28, 24);

    pL->addWidget(makeLine());

    // Field nama pemilik
    pL->addWidget(makeCaption("NAMA PEMILIK"));
    inputNama = new QLineEdit;
    inputNama->setPlaceholderText("Masukkan nama lengkap");
    inputNama->setStyleSheet(SS_INPUT);
    inputNama->setFixedHeight(42);

    cNomor->addWidget(inputNomor);

    cJenis->setSpacing(6);
    cJenis->addWidget(makeCaption("JENIS KENDARAAN"));

    scrollArea->setWidget(innerWidget);
    scrollArea->setWidgetResizable(true);

    QVBoxLayout *inputPageLayout = new QVBoxLayout(inputPage);
    inputPageLayout->setContentsMargins(0, 0, 0, 0);
    inputPageLayout->addWidget(scrollArea);

/**
 * @brief MainWindow::showInputPage
 * Menampilkan halaman Input Data dan mengaktifkan tombol navigasinya di sidebar.
 */
void MainWindow::showInputPage()
{
    contentStack->setCurrentWidget(inputPage);
    btnInputData->setStyleSheet(SS_NAV_BTN_ACTIVE);
    btnRiwayat->setStyleSheet(SS_NAV_BTN);
}

/**
 * @brief MainWindow::showRiwayatPage
 * Menampilkan halaman Riwayat. Data terbaru selalu dikirim ulang ke widget
 * Riwayat sebelum halaman ditampilkan agar tabel selalu up-to-date.
 */
void MainWindow::showRiwayatPage()
{
    riwayatPage->refresh(riwayatData);
    contentStack->setCurrentWidget(riwayatPage);
    btnRiwayat->setStyleSheet(SS_NAV_BTN_ACTIVE);
    btnInputData->setStyleSheet(SS_NAV_BTN);
}

/**
 * @brief MainWindow::kembaliKeInput
 * Slot yang dipanggil ketika tombol "Kembali" di halaman Hasil ditekan.
 */
void MainWindow::kembaliKeInput()
{
    showInputPage();
}

// ── Slot Detail ───────────────────────────────────────────────

/**
 * @brief MainWindow::bukaDetailRiwayat
 * Dipanggil saat user mengklik salah satu baris di tabel Riwayat.
 * Mengambil data dari riwayatData berdasarkan index dan mengirimnya
 * ke halaman Detail untuk ditampilkan.
 * @param index Index record di dalam QList riwayatData.
 */
void MainWindow::bukaDetailRiwayat(int index)
{
    if (index < 0 || index >= riwayatData.size()) return;
    detailPage->tampilkan(riwayatData[index], index);
    contentStack->setCurrentWidget(detailPage);
    btnRiwayat->setStyleSheet(SS_NAV_BTN_ACTIVE);
    btnInputData->setStyleSheet(SS_NAV_BTN);
}

/**
 * @brief MainWindow::kembaliKeRiwayatDariDetail
 * Slot yang dipanggil ketika tombol "Kembali ke Riwayat" di halaman Detail ditekan.
 */
void MainWindow::kembaliKeRiwayatDariDetail()
{
    showRiwayatPage();
}

/**
 * @brief MainWindow::hapusRiwayat
 * Menghapus record pada index tertentu dari QList, menyimpan perubahan
 * ke file, lalu kembali ke halaman Riwayat.
 * @param index Index record yang akan dihapus di dalam riwayatData.
 */
void MainWindow::hapusRiwayat(int index)
{
    if (index < 0 || index >= riwayatData.size()) return;
    riwayatData.removeAt(index);
    Save::simpanKeFile(riwayatData);
    QMessageBox::information(this, "Berhasil", "Data berhasil dihapus.");
    showRiwayatPage();
}

/**
 * @brief MainWindow::updateRiwayat
 * Memperbarui record pada index tertentu dengan data hasil edit,
 * menyimpan ke file, lalu memperbarui tampilan halaman Detail.
 * @param index Index record yang akan diperbarui di dalam riwayatData.
 * @param data  Data kendaraan baru hasil dari form edit di halaman Detail.
 */
void MainWindow::updateRiwayat(int index, DataKendaraan data)
{
    if (index < 0 || index >= riwayatData.size()) return;
    riwayatData[index] = data;
    Save::simpanKeFile(riwayatData);
    QMessageBox::information(this, "Berhasil", "Data berhasil diperbarui.");
    detailPage->tampilkan(riwayatData[index], index);
}

// ── Slot hitung pajak ─────────────────────────────────────────

/**
 * @brief MainWindow::hitungPajak
 * Dipanggil saat tombol "HITUNG PAJAK" ditekan.
 * Melakukan validasi input satu per satu, mendelegasikan perhitungan
 * ke kelas HitungPajak, lalu menampilkan hasilnya di halaman Hasil.
 */
void MainWindow::hitungPajak()
{
    // Validasi field wajib satu per satu agar pesan error spesifik
    if (inputNama->text().trimmed().isEmpty()) {
        QMessageBox::warning(this, "Validasi", "Nama pemilik harus diisi!"); return;
    }
    if (inputNomor->text().trimmed().isEmpty()) {
        QMessageBox::warning(this, "Validasi", "Nomor polisi harus diisi!"); return;
    }
    if (comboJenis->currentIndex() == 0) {
        QMessageBox::warning(this, "Validasi", "Pilih jenis kendaraan terlebih dahulu!"); return;
    }
    if (inputTahun->text().trimmed().isEmpty()) {
        QMessageBox::warning(this, "Validasi", "Tahun pembuatan harus diisi!"); return;
    }

    // Bersihkan format Rupiah jika user sudah mengetik dengan titik atau "Rp"
    QString raw = inputNJKB->text().trimmed();
    raw.remove("Rp").remove(".").remove(",").remove(" ");
    double njkb = raw.toDouble();
    if (njkb <= 0) {
        QMessageBox::warning(this, "Validasi", "NJKB harus diisi dengan nilai yang valid!"); return;
    }

    QString jenis = comboJenis->currentText();

    // Delegasikan perhitungan ke kelas HitungPajak (prinsip Single Responsibility)
    double pkb   = HitungPajak::hitungPKB(njkb, jenis);
    double swdkl = HitungPajak::hitungSWDKLLJ(jenis);
    double total = pkb + swdkl;

    // Simpan ke currentData sebagai buffer sebelum user konfirmasi simpan
    currentData = {
        inputNama->text().trimmed(),
        inputAlamat->toPlainText().trimmed(),
        inputNomor->text().trimmed().toUpper(),
        jenis,
        inputTahun->text().toInt(),
        njkb, pkb, swdkl, total,
        QDateTime::currentDateTime().toString("dd/MM/yyyy")
    };

    // Kirim data yang sudah diformat ke widget Hasil lalu tampilkan halamannya
    hasilPage->tampilkanData(
        currentData,
        Save::formatRupiah(pkb),
        Save::formatRupiah(swdkl),
        Save::formatRupiah(total)
        );

    contentStack->setCurrentWidget(hasilPage);
    btnInputData->setStyleSheet(SS_NAV_BTN_ACTIVE);
    btnRiwayat->setStyleSheet(SS_NAV_BTN);
}

// ── Slot simpan ke riwayat ────────────────────────────────────

/**
 * @brief MainWindow::simpanKeRiwayat
 * Dipanggil saat tombol "Simpan ke Riwayat" di halaman Hasil ditekan.
 * Menambahkan currentData ke QList dan langsung menyimpan ke file
 * agar data tidak hilang saat aplikasi ditutup.
 */
void MainWindow::simpanKeRiwayat()
{
    Save::tambahRiwayat(riwayatData, currentData);
    Save::simpanKeFile(riwayatData);
    QMessageBox::information(this, "Berhasil", "Data berhasil disimpan ke riwayat!");
    showRiwayatPage();
}
