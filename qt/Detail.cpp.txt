static QFrame* makeCard() {
    QFrame *f = new QFrame;
    f->setStyleSheet("QFrame{background:#ffffff;border:1px solid #e2e8f0;border-radius:12px;}");
    auto *sh = new QGraphicsDropShadowEffect;
    sh->setBlurRadius(12); sh->setColor(QColor(0,0,0,15)); sh->setOffset(0,2);
    f->setGraphicsEffect(sh);
    return f;
}
static QFrame* makeLine() {
    QFrame *f = new QFrame; f->setFrameShape(QFrame::HLine);
    f->setFixedHeight(1); f->setStyleSheet("background:#e2e8f0;border:none;");
    return f;
}
static QLabel* makeCaption(const QString &t) {
    QLabel *l = new QLabel(t);
    l->setStyleSheet("font-size:11px;font-weight:bold;color:#64748b;"
                     "letter-spacing:0.8px;background:transparent;border:none;");
    return l;
}
static QLabel* makeValueLabel() {
    QLabel *l = new QLabel("...");
    l->setStyleSheet("font-size:14px;color:#1e293b;background:transparent;border:none;");
    l->setWordWrap(true);
    return l;
}
static QLabel* makeBigValue() {
    QLabel *l = new QLabel("Rp 0");
    l->setStyleSheet("font-size:26px;font-weight:bold;color:#2563eb;"
                     "background:transparent;border:none;");
    return l;
}

// Public: tampilkan data ke halaman ini
void Detail::tampilkan(const DataKendaraan &data, int index)
{
    currentData  = data;
    currentIndex = index;
    isiViewMode(data);
    modeStack->setCurrentWidget(viewWidget);
}

void Detail::isiViewMode(const DataKendaraan &d)
{
    vNama   ->setText(d.namaPemilik);
    vAlamat ->setText(d.alamat.isEmpty() ? "-" : d.alamat);
    vNomor  ->setText(d.nomorPolisi);
    vJenis  ->setText(d.jenisKendaraan);
    vTahun  ->setText(QString::number(d.tahunPembuatan));
    vTanggal->setText(d.tanggalHitung);
    vNJKB   ->setText(Save::formatRupiah(d.njkb));
    vPKB    ->setText(Save::formatRupiah(d.pkb));
    vSWD    ->setText(Save::formatRupiah(d.swdkllj));
    vTotal  ->setText(Save::formatRupiah(d.totalPajak));
}

void Detail::isiEditMode(const DataKendaraan &d)
{
    eNama  ->setText(d.namaPemilik);
    eAlamat->setPlainText(d.alamat);
    eNomor ->setText(d.nomorPolisi);
    eTahun ->setText(QString::number(d.tahunPembuatan));
    eNJKB  ->setText(QString::number(static_cast<long long>(d.njkb)));

    // Pilih item combo yang sesuai
    int idx = eJenis->findText(d.jenisKendaraan);
    if (idx >= 0) eJenis->setCurrentIndex(idx);
}

// Slot
void Detail::onEdit()
{
    isiEditMode(currentData);
    modeStack->setCurrentWidget(editWidget);
}

void Detail::onBatalEdit()
{
    modeStack->setCurrentWidget(viewWidget);
}

void Detail::onHapus()
{
    int konfirmasi = QMessageBox::question(
        this, "Konfirmasi Hapus",
        QString("Apakah kamu yakin ingin menghapus data atas nama\n\"%1\"?\n"
                "Tindakan ini tidak dapat dibatalkan.")
            .arg(currentData.namaPemilik),
        QMessageBox::Yes | QMessageBox::No,
        QMessageBox::No
        );
    if (konfirmasi == QMessageBox::Yes)
        emit hapusDiminta(currentIndex);
}

void Detail::onSimpanEdit()
{
    // Validasi input edit
    if (eNama->text().trimmed().isEmpty()) {
        QMessageBox::warning(this, "Validasi", "Nama pemilik tidak boleh kosong!"); return;
    }
    if (eNomor->text().trimmed().isEmpty()) {
        QMessageBox::warning(this, "Validasi", "Nomor polisi tidak boleh kosong!"); return;
    }
    QString rawNJKB = eNJKB->text().trimmed();
    rawNJKB.remove("Rp").remove(".").remove(",").remove(" ");
    double njkb = rawNJKB.toDouble();
    if (njkb <= 0) {
        QMessageBox::warning(this, "Validasi", "NJKB harus diisi dengan nilai yang valid!"); return;
    }

    // Hitung ulang pajak berdasarkan data yang diedit
    QString jenis = eJenis->currentText();
    double pkb    = HitungPajak::hitungPKB(njkb, jenis);
    double swdkl  = HitungPajak::hitungSWDKLLJ(jenis);

    // Susun DataKendaraan baru
    DataKendaraan updated;
    updated.namaPemilik    = eNama->text().trimmed();
    updated.alamat         = eAlamat->toPlainText().trimmed();
    updated.nomorPolisi    = eNomor->text().trimmed().toUpper();
    updated.jenisKendaraan = jenis;
    updated.tahunPembuatan = eTahun->text().toInt();
    updated.njkb           = njkb;
    updated.pkb            = pkb;
    updated.swdkllj        = swdkl;
    updated.totalPajak     = pkb + swdkl;
    updated.tanggalHitung  = currentData.tanggalHitung; // tanggal asli tetap

    currentData = updated;
    isiViewMode(updated);
    modeStack->setCurrentWidget(viewWidget);

    emit editDisimpan(currentIndex, updated);
}
