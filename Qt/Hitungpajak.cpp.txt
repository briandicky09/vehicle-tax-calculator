#include "Hitungpajak.h"

// PKB kepemilikan pertama = 1,5% NJKB
double HitungPajak::hitungPKB(double njkb, const QString &jenis)
{
    Q_UNUSED(jenis)
    return njkb * 0.015;
}

// Tarif SWDKLLJ
double HitungPajak::hitungSWDKLLJ(const QString &jenis)
{
    if (jenis == "Sepeda Motor")           return 35000.0;
    if (jenis.contains("Mobil Penumpang")) return 143000.0;
    if (jenis == "Mobil Barang")           return 163000.0;
    if (jenis.contains("Bus"))             return 163000.0;
    return 143000.0; // default: kendaraan penumpang
}
